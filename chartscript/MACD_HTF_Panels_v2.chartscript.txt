// Port notes (what differs from the Pine v6 original):
//   - HTF panels are drawn in THIS indicator's pane, to the right of the last bar.
//     ChartScript documents force_overlay only for plots, shapes, bgcolor, tables and polylines,
//     not for line.new / label.new / box.new, so the panels cannot be pinned to the price pane.
//     The panel "size unit" is now the chart MACD's recent peak amplitude (not ATR), and the
//     vertical centre is measured from the zero line (not from close).
//   - request.security returns plain series (a tuple), never arrays. The rolling window of HTF
//     bars is rebuilt on the chart side in a ring buffer, one entry per HTF bar.
//   - Only lookahead_off is used. "Closed HTF bars only" mode drops the newest HTF entry while
//     that HTF bar has not closed before the current chart bar opened, so nothing reads the future.
//   - Zone levels are plots (hline levels are documented as const), so line style and width for
//     each zone are set from the Style tab instead of from inputs.
//   - Every frame redraw first deletes all of this script's lines, labels and boxes, so the
//     panels never pile up on live ticks regardless of how drawings are rolled back.

int KEEP_BARS = panel_bars + 2    // skip slot + panel bars + one extra for prev-value colouring

// ───────────────────────────── Types ─────────────────────────────
// Rolling window of one higher timeframe, newest HTF bar at index 0.
type HtfBuf
    array<int>   t
    array<int>   tc
    array<float> o
    array<float> h
    array<float> l
    array<float> c
    array<float> m
    array<float> s
    array<float> x

// ───────────────────────────── Helpers ─────────────────────────────
newBuf() =>
    HtfBuf.new(array.new<int>(), array.new<int>(), array.new<float>(), array.new<float>(), array.new<float>(), array.new<float>(), array.new<float>(), array.new<float>(), array.new<float>())

// A new HTF bar (new open time) is prepended; the same HTF bar overwrites slot 0, so a
// forming HTF bar keeps updating in place on live ticks.
method feed(HtfBuf this, int t, int tc, float o, float h, float l, float c, float m, float s, float x, int keep) =>
    if not na(t)
        bool isNew = this.t.size() == 0 or t != this.t.get(0)
        if isNew
            this.t.unshift(t)
            this.tc.unshift(tc)
            this.o.unshift(o)
            this.h.unshift(h)
            this.l.unshift(l)
            this.c.unshift(c)
            this.m.unshift(m)
            this.s.unshift(s)
            this.x.unshift(x)
            if this.t.size() > keep
                this.t.pop()
                this.tc.pop()
                this.o.pop()
                this.h.pop()
                this.l.pop()
                this.c.pop()
                this.m.pop()
                this.s.pop()
                this.x.pop()
        else
            this.tc.set(0, tc)
            this.o.set(0, o)
            this.h.set(0, h)
            this.l.set(0, l)
            this.c.set(0, c)
            this.m.set(0, m)
            this.s.set(0, s)
            this.x.set(0, x)
    this.t.size()

get_table_position() =>
    switch table_position
        "Top Left"     => position.top_left
        "Top Right"    => position.top_right
        "Bottom Left"  => position.bottom_left
        "Bottom Right" => position.bottom_right
        "Middle Left"  => position.middle_left
        "Middle Right" => position.middle_right
        => position.bottom_right

f_at(array<float> a, int i) =>
    float r = na
    if not na(a)
        if i >= 0 and i < a.size()
            r := a.get(i)
    r

f_fmtVal(float v) =>
    string out = "—"
    if not na(v)
        float a = math.abs(v)
        string fmt = a >= 100 ? "#,##0.0" : a >= 1 ? "0.00" : a >= 0.01 ? "0.0000" : "0.000000"
        out := str.tostring(v, fmt)
    out

f_histColor(float v, float vPrev) =>
    bool rising = not na(vPrev) and v >= vPrev
    v == 0 ? color.gray : v > 0 ? (rising ? HIST_POS_UP : HIST_POS_DOWN) : (rising ? HIST_NEG_UP : HIST_NEG_DOWN)

f_map(float v, float lo, float span, float yBot, float yTop) =>
    yBot + (v - lo) / span * (yTop - yBot)

// HTF ladder: 5, 15, 30, 60, 240, D, W, M, 3M, 6M, 12M
f_ladderTf(int i) =>
    switch i
        0 => "5"
        1 => "15"
        2 => "30"
        3 => "60"
        4 => "240"
        5 => "D"
        6 => "W"
        7 => "M"
        8 => "3M"
        9 => "6M"
        => "12M"

// First ladder entry strictly above the chart timeframe.
f_ladderStart() =>
    int sec = timeframe.in_seconds(timeframe.period)
    switch
        na(sec) or sec < 300               => 0
        sec < 900                          => 1
        sec < 1800                         => 2
        sec < 3600                         => 3
        sec < 14400                        => 4
        sec < 86400                        => 5
        sec < timeframe.in_seconds("W")    => 6
        sec < timeframe.in_seconds("M")    => 7
        not allow_multi_month              => 7
        sec < timeframe.in_seconds("3M")   => 8
        sec < timeframe.in_seconds("6M")   => 9
        => 10

f_autoHtf(int step) =>
    int maxIdx = allow_multi_month ? 10 : 7
    f_ladderTf(math.min(f_ladderStart() + step - 1, maxIdx))

// "60" -> "1H", "360" -> "6H", "45" -> "45m", "30S" -> "30s", D/W/M unchanged.
// Flags a manual TF lower than the chart TF with "⚠".
f_tfLabel(string tf) =>
    int sec = timeframe.in_seconds(tf)
    string base = sec < 60 ? str.tostring(sec) + "s" : sec < 86400 ? (sec % 3600 == 0 ? str.tostring(math.floor(sec / 3600)) + "H" : str.tostring(math.floor(sec / 60)) + "m") : tf
    sec < timeframe.in_seconds(timeframe.period) ? base + " ⚠" : base

f_title(string lbl, int bias) =>
    lbl + (bias > 0 ? "  ▲" : bias < 0 ? "  ▼" : "")

// Evaluated inside request.security on the HTF bars: one row of OHLC + MACD per HTF bar.
htfData() =>
    [hm, hs, hx] = ta.macd(close, macd_fast, macd_slow, macd_signal)
    [time, time_close, open, high, low, close, hm, hs, hx]

// Index of the newest usable HTF bar. In closed-bar mode the newest entry is skipped while
// its HTF bar has not closed before the current chart bar opened.
f_skip(HtfBuf b, string tf) =>
    int sk = 0
    if confirmed_only and b.t.size() > 0
        int tc0  = b.tc.get(0)
        int endT = na(tc0) ? b.t.get(0) + timeframe.in_seconds(tf) * 1000 : tc0
        sk := endT > time ? 1 : 0
    sk

// Cross / bias state from the newest usable bar (index sk) vs the one before it.
f_state(HtfBuf b, int sk) =>
    float mc = f_at(b.m, sk)
    float sc = f_at(b.s, sk)
    float mp = f_at(b.m, sk + 1)
    float sp = f_at(b.s, sk + 1)
    bool valid = not na(mc) and not na(sc) and not na(mp) and not na(sp)
    bool bullX = valid and mp - sp <= 0 and mc - sc > 0
    bool bearX = valid and mp - sp >= 0 and mc - sc < 0
    int bias = na(mc) or na(sc) ? 0 : mc > sc ? 1 : mc < sc ? -1 : 0
    [bullX, bearX, bias]

// ───────────────────────────── Panel renderer ─────────────────────────────
// Draws one HTF panel. Called only on the last bar, after all previous drawings were deleted.
renderPanel(HtfBuf b, int sk, string title, float yC, int xOff, float unit, float hUnits, int maxBars, int w, bool titleRight) =>
    int n = math.min(b.o.size() - sk, maxBars)
    if n > 1
        // only the usable window (excludes the developing bar in closed-bar mode)
        oS = b.o.slice(sk, sk + n)
        hS = b.h.slice(sk, sk + n)
        lS = b.l.slice(sk, sk + n)
        cS = b.c.slice(sk, sk + n)
        mS = b.m.slice(sk, sk + n)
        sS = b.s.slice(sk, sk + n)
        xS = b.x.slice(sk, sk + n)

        // geometry (pane units: MACD value space, centred on yC size units)
        float H    = hUnits * unit
        float yTop = yC * unit + H / 2.0
        float yBot = yTop - H
        float cTop = yTop
        float cBot = yTop - H * 0.40
        float mTop = show_panel_candles ? cBot - H * 0.05 : yTop
        float mBot = yBot

        // price range
        float pLo   = lS.min()
        float pHi   = hS.max()
        float pSpan = na(pLo) or na(pHi) or pHi - pLo == 0 ? 1.0 : pHi - pLo

        // MACD range always includes zero, so histogram columns stay inside the band
        float mLo   = math.min(0.0, nz(mS.min()), nz(sS.min()), nz(xS.min()))
        float mHi   = math.max(0.0, nz(mS.max()), nz(sS.max()), nz(xS.max()))
        float mSpan = mHi - mLo == 0 ? 1.0 : mHi - mLo

        int   x0    = bar_index + xOff
        int   xEnd  = x0 + (n - 1) * w
        float yZero = f_map(0.0, mLo, mSpan, mBot, mTop)
        int   hw    = w >= 2 ? 3 : 2

        if show_frame
            box.new(x0 - 1, yTop, xEnd + 1, yBot, border_color=panel_border_col, border_width=1, bgcolor=panel_bg_col)

        if show_zero_line
            line.new(x0, yZero, xEnd, yZero, color=color.new(color.gray, 40), style=line.style_dotted, width=1)

        for idx = 0 to n - 1
            int j = n - 1 - idx          // arrays are newest-first; draw left-to-right
            int x = x0 + idx * w

            if show_panel_candles
                float o = oS.get(j)
                float c = cS.get(j)
                float hh = hS.get(j)
                float ll = lS.get(j)
                if not na(o) and not na(c) and not na(hh) and not na(ll)
                    color cc = color.new(o > c ? panel_bear_col : panel_bull_col, panel_candle_transp)
                    line.new(x, f_map(hh, pLo, pSpan, cBot, cTop), x, f_map(ll, pLo, pSpan, cBot, cTop), color=cc, width=1)
                    line.new(x, f_map(o, pLo, pSpan, cBot, cTop), x, f_map(c, pLo, pSpan, cBot, cTop), color=cc, width=3)

            float v3 = xS.get(j)
            if not na(v3)
                float v3p = f_at(b.x, sk + j + 1)
                line.new(x, yZero, x, f_map(v3, mLo, mSpan, mBot, mTop), color=f_histColor(v3, v3p), width=hw)

            if idx < n - 1
                int   jn = j - 1
                int   xn = x + w
                float mA = mS.get(j)
                float mB = mS.get(jn)
                float sA = sS.get(j)
                float sB = sS.get(jn)
                if not na(mA) and not na(mB)
                    line.new(x, f_map(mA, mLo, mSpan, mBot, mTop), xn, f_map(mB, mLo, mSpan, mBot, mTop), color=MACD_LINE_COLOR, width=2)
                if not na(sA) and not na(sB)
                    line.new(x, f_map(sA, mLo, mSpan, mBot, mTop), xn, f_map(sB, mLo, mSpan, mBot, mTop), color=MACD_SIGNAL_COLOR, width=1)
                if show_cross_marks and not na(mA) and not na(mB) and not na(sA) and not na(sB)
                    float dPrev = mA - sA
                    float dCur  = mB - sB
                    if dPrev <= 0 and dCur > 0
                        label.new(xn, f_map(mB, mLo, mSpan, mBot, mTop), "", color=HIST_POS_UP, style=label.style_circle, size=size.tiny)
                    else if dPrev >= 0 and dCur < 0
                        label.new(xn, f_map(mB, mLo, mSpan, mBot, mTop), "", color=HIST_NEG_DOWN, style=label.style_circle, size=size.tiny)

        if show_panel_labels
            // right-column titles sit to the right of their panel so they never cover the left column
            if titleRight
                label.new(xEnd + 1 + panel_label_offset, (yTop + yBot) / 2.0, title, color=color.new(color.black, 100), style=label.style_label_left, textcolor=chart.fg_color, size=size.normal)
            else
                label.new(x0 - 1 - panel_label_offset, (yTop + yBot) / 2.0, title, color=color.new(color.black, 100), style=label.style_label_right, textcolor=chart.fg_color, size=size.normal)
    n

_n3 = buf3.feed(q3t, q3tc, q3o, q3h, q3l, q3c, q3m, q3s, q3x, KEEP_BARS)
_n4 = buf4.feed(q4t, q4tc, q4o, q4h, q4l, q4c, q4m, q4s, q4x, KEEP_BARS)

sk1 = f_skip(buf1, htf1_tf)
sk2 = f_skip(buf2, htf2_tf)
sk3 = f_skip(buf3, htf3_tf)
sk4 = f_skip(buf4, htf4_tf)

bool a2 = panel_count >= 2
bool a3 = panel_count >= 3
bool a4 = panel_count >= 4

[bx1, sx1, b1] = f_state(buf1, sk1)
[bx2, sx2, b2] = f_state(buf2, sk2)
[bx3, sx3, b3] = f_state(buf3, sk3)
[bx4, sx4, b4] = f_state(buf4, sk4)

int bullCnt = (b1 > 0 ? 1 : 0) + (a2 and b2 > 0 ? 1 : 0) + (a3 and b3 > 0 ? 1 : 0) + (a4 and b4 > 0 ? 1 : 0)
int bearCnt = (b1 < 0 ? 1 : 0) + (a2 and b2 < 0 ? 1 : 0) + (a3 and b3 < 0 ? 1 : 0) + (a4 and b4 < 0 ? 1 : 0)
bool allBull = bullCnt == panel_count
bool allBear = bearCnt == panel_count

// ───────────────────────────── Layout / budgets ─────────────────────────────
// 1 size unit = peak |MACD|, |Signal| or |Hist| of the chart timeframe over the lookback.
float panelUnit = ta.highest(math.max(math.abs(nz(curMacdLine)), math.abs(nz(curMacdSignal)), math.abs(nz(curMacdHist))), panel_unit_len)

// 500-line cap: per panel = 1 zero line + per bar (hist + MACD + signal [+ 2 candle lines])
int cost_per_bar = 3 + (show_panel_candles ? 2 : 0)
int line_budget  = 490 - panel_count
int render_bars  = math.max(2, math.min(panel_bars, math.floor(line_budget / (panel_count * cost_per_bar))))

// keep drawings within ~500 bars past the last bar
int max_w    = panel_count == 4 ? math.floor((493 - panel_xoffset - panel_col_gap) / (2 * render_bars)) : math.floor((493 - panel_xoffset) / render_bars)
int candle_w = math.max(1, math.min(panel_candle_width, max_w))

// Every layout fills the same vertical budget as the 3-panel stack.
float stack_budget = 3.0 * panel_height + 2.0 * panel_gap
float auto_panel_height = switch panel_count
    1 => stack_budget
    2 => (stack_budget - panel_gap) / 2.0
    4 => (stack_budget - panel_gap) / 2.0
    => panel_height
float eff_height = panel_size_mode == "Manual" ? panel_height * panel_manual_scale : auto_panel_height

float slot_spacing = (panel_count == 2 or panel_count == 4) ? (eff_height + panel_gap) / 2.0 : (eff_height + panel_gap)
float y_top = panel_center_y + slot_spacing
float y_mid = panel_center_y
float y_bot = panel_center_y - slot_spacing

int col1_x = panel_xoffset
int col2_x = panel_xoffset + render_bars * candle_w + panel_col_gap

// slot assignment
float y1 = panel_count == 1 ? y_mid : y_top
float y2 = panel_count == 3 ? y_mid : y_bot
float y3 = panel_count == 3 ? y_bot : y_top
float y4 = y_bot
int   x3 = panel_count == 4 ? col2_x : col1_x

if barstate.islast
    // Remove everything drawn by the previous run (last tick), then redraw.
    oldLines = array.copy(line.all)
    for ln in oldLines
        line.delete(ln)
    oldLabels = array.copy(label.all)
    for lb in oldLabels
        label.delete(lb)
    oldBoxes = array.copy(box.all)
    for bx in oldBoxes
        box.delete(bx)

    if show_panels and not na(panelUnit) and panelUnit > 0
        _r1 = renderPanel(buf1, sk1, f_title(htf1_lbl, b1), y1, col1_x, panelUnit, eff_height, render_bars, candle_w, false)
        if a2
            _r2 = renderPanel(buf2, sk2, f_title(htf2_lbl, b2), y2, col1_x, panelUnit, eff_height, render_bars, candle_w, false)
        if a3
            _r3 = renderPanel(buf3, sk3, f_title(htf3_lbl, b3), y3, x3, panelUnit, eff_height, render_bars, candle_w, panel_count == 4)
        if a4
            _r4 = renderPanel(buf4, sk4, f_title(htf4_lbl, b4), y4, col2_x, panelUnit, eff_height, render_bars, candle_w, true)

// ───────────────────────────── Signal table ─────────────────────────────
f_row(table t, int r, string lbl, HtfBuf b, int sk, int bias, color bg) =>
    float hv = f_at(b.x, sk)
    table.cell(t, 0, r, lbl, text_color=color.silver, text_size=size.small, bgcolor=bg)
    table.cell(t, 1, r, f_fmtVal(f_at(b.m, sk)), text_color=color.white, text_size=size.small, bgcolor=bg)
    table.cell(t, 2, r, f_fmtVal(f_at(b.s, sk)), text_color=color.white, text_size=size.small, bgcolor=bg)
    table.cell(t, 3, r, f_fmtVal(hv), text_color=na(hv) ? color.silver : f_histColor(hv, f_at(b.x, sk + 1)), text_size=size.small, bgcolor=bg)
    table.cell(t, 4, r, bias > 0 ? "▲" : bias < 0 ? "▼" : "•", text_color=bias > 0 ? HIST_POS_UP : bias < 0 ? HIST_NEG_DOWN : color.gray, text_size=size.small, bgcolor=bg)
    r

        table.cell(sigTable, 3, r, "", bgcolor=cbg)
        table.cell(sigTable, 4, r, allBull ? "ALL ▲" : allBear ? "ALL ▼" : "MIXED", text_color=color.white, text_size=size.small, bgcolor=cbg)
